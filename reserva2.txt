#include <iostream>

namespace {
    ColorData cor_global;
    std::vector<LightData> LightList;
    std::vector<ObjectData> objetos_validos;
    Ponto CameraPos;
    
    // Função de tonemapping simples (reinhard)

    bool notTIR(double n_i, double n_t, const Vetor& N, const Vetor& d_neg) {
        double cos_i = N.dot(d_neg);           // cos(θi)
        double ratio = n_i / n_t;
        double discriminant = 1.0 - (ratio * ratio) * (1.0 - cos_i * cos_i);
        return discriminant >= 0.0;            // false = TIR ocorre
    }

    Vetor refractDirection(double n_i, double n_t, const Vetor& N, const Vetor& d_neg) {
        double ratio = n_i / n_t;
        // d_neg é -direcao, então o raio incidente real I é:
        Vetor I = d_neg * (-1.0); 
        
        double cos_i = N.dot(d_neg); 
        double disc = 1.0 - (ratio * ratio) * (1.0 - cos_i * cos_i);
        double cos_t = std::sqrt(disc);

        // Fórmula canônica vetorial de Snell
        return (I * ratio) + (N * (ratio * cos_i - cos_t));
    }

    std::array<double, 3> RayTracer(const Vetor& direcao, const Ponto& origem , int iteracao){
        std::array<double, 3> arr = {0, 0, 0};
        if(iteracao >= limite_recursao){
            return arr;
        }
        
        double closest_t = infinity();
        const ObjectData* hit_obj = nullptr;
        Vetor hit_normal;
        
        for(const auto& objeto : objetos_validos){
            HitResult hr = intersect_object(objeto, origem, direcao);
            if(hr.t < closest_t){
                closest_t  = hr.t;
                hit_obj    = &objeto;
                hit_normal = hr.normal;
            }
        }
        
        if(hit_obj){
            Ponto P = origem + (direcao * closest_t);
            bool entrando = direcao.dot(hit_normal) < 0.0;
            
            // Mantém a normal orientada separada para não alterar a hit_normal geométrica original
            Vetor normal_orientada = entrando ? hit_normal : hit_normal * (-1.0);
            
            auto [cor_r, cor_g, cor_b] = calcular_cor_phong(
                P, normal_orientada, direcao, *hit_obj, cor_global, CameraPos, LightList, objetos_validos, intersect_object);
            
            MaterialData Material = hit_obj->material;
            
            // FATOR DE TRANSPARÊNCIA CONCEITUAL: Se d = 0.02, a transparência é 0.98
            double transparencia = 1.0 - Material.d;
            
            // A parte OPACA do objeto absorve a cor de Phong local
            arr[0] = cor_r * Material.d;
            arr[1] = cor_g * Material.d;
            arr[2] = cor_b * Material.d;
            
            // --- CÁLCULO DE REFLEXÃO ---
            if(Material.kr.r > 0.0 || Material.kr.g > 0.0 || Material.kr.b > 0.0){
                Ponto P_refletido = P + (normal_orientada * 1e-4);
                Vetor refletido = direcao - normal_orientada * (2.0 * direcao.dot(normal_orientada));
                std::array<double, 3> cor_refletido = RayTracer(refletido, P_refletido, iteracao + 1);
                
            arr[0] += Material.d *Material.kr.r * cor_refletido[0];

            arr[1] += Material.d *Material.kr.g * cor_refletido[1];

            arr[2] += Material.d *Material.kr.b * cor_refletido[2];
            }
            
            // --- CÁLCULO DE REFRAÇÃO ---
            if(Material.kt.r > 0 || Material.kt.g > 0 || Material.kt.b > 0){
                double n_i, n_t;
                if(entrando){
                    n_i = 1.0; n_t = Material.ni;
                }
                else{
                    n_i = Material.ni; n_t = 1.0;
                }
                
                if(notTIR(n_i, n_t, normal_orientada , -direcao)){
                    Ponto P_refratado = P - (normal_orientada * 1e-4);
                    Vetor refratado = refractDirection(n_i, n_t, normal_orientada , -direcao).normalize();
                    std::array<double, 3> cor_refratado = RayTracer(refratado, P_refratado, iteracao + 1);
                    
                    // CORREÇÃO: A luz refratada passa pela fração transparente e é filtrada pela cor intrínseca do objeto
                    arr[0] += transparencia * Material.kt.r * Material.color.r * cor_refratado[0];
                    arr[1] += transparencia * Material.kt.g * Material.color.g * cor_refratado[1];
                    arr[2] += transparencia * Material.kt.b * Material.color.b * cor_refratado[2];
                } 
                else {
                    // Reflexão Interna Total (TIR)
                    Ponto P_refletido_interno = P + (normal_orientada * 1e-4);
                    Vetor refletido_interno = direcao - normal_orientada * (2.0 * direcao.dot(normal_orientada));
                    std::array<double, 3> cor_tir = RayTracer(refletido_interno, P_refletido_interno, iteracao + 1);
                    
                    arr[0] += cor_tir[0];
                    arr[1] += cor_tir[1];
                    arr[2] += cor_tir[2];
                }
            }
        }
        return arr;
    }
}

void Trace(const CenaProcessada& dados, const Camera& cam, const SceneData& scene){
    std::vector<std::array<int, 3>> image_buffer(cam.hres * cam.vres);
    objetos_validos = dados.valid_objects;
    LightList = scene.lightList;
    cor_global = scene.globalLight.color;
    CameraPos = scene.camera.lookfrom;
    
    for(int j = 0; j < cam.vres; j ++){
        std::cerr << "Linha " << j << '/' << cam.vres << '\r' << std::flush;
        for(int i = 0; i < cam.hres; i++){
            int pixel_index = j * cam.hres + i;
            std::array<double, 3> cor_acumulada = {0.0, 0.0, 0.0};
            
            Vetor ray = cam.getRayDirection((double)i , (double)j);
            auto c = RayTracer(ray, CameraPos, 1);
            
            cor_acumulada[0] += c[0];
            cor_acumulada[1] += c[1];
            cor_acumulada[2] += c[2];
            
            image_buffer[pixel_index] = {
                std::min(255, (int)(255.999 * cor_acumulada[0])),
                std::min(255, (int)(255.999 * cor_acumulada[1])),
                std::min(255, (int)(255.999 * cor_acumulada[2]))
            };
        }
    }
    for (const auto& pixel : image_buffer) {
        std::cout << pixel[0] << ' ' << pixel[1] << ' ' << pixel[2] << '\n';
    }
}
